  function change3DPos(pos, btn) {
    // 1. Update UI
    document.querySelectorAll('.pos-card').forEach(b => b.classList.remove('active'));
    btn.classList.add('active');

    // 2. Reset Stage
    const stage = document.getElementById('stage3D');
    stage.classList.remove('latching', 'gaping');
    
    // 3. Update Text
    const status = document.getElementById('latchStatus');
    status.className = 'latch-feedback-box';
    status.innerHTML = `<i class="fas fa-info-circle" style="margin-right:10px;"></i><span>Aligned for <strong>${pos}</strong> hold. Ready to latch.</span>`;
    
    // 4. Move Baby to Starting Position
    stage.setAttribute('data-pos', pos);
  }

  function animateDeepLatch() {
    const stage = document.getElementById('stage3D');
    const status = document.getElementById('latchStatus');
    const tongue = document.getElementById('baby-tongue');
    
    // RESET
    stage.classList.remove('latching', 'gaping');
    tongue.style.opacity = '0';
    void stage.offsetWidth; // Force reflow
    
    // PHASE 1: ALIGNMENT (Nose to Nipple)
    status.className = 'latch-feedback-box active';
    status.innerHTML = '<span><strong>Step 1:</strong> Aligning Nose to Nipple... (Chin tucked)</span>';
    
    setTimeout(() => {
      // PHASE 2: THE GAPE (Head tilts back, Mouth opens wide)
      stage.classList.add('gaping');
      status.innerHTML = '<span><strong>Step 2:</strong> Waiting for wide gape (160°)...</span>';
      
      setTimeout(() => {
        // PHASE 3: ASYMMETRIC LATCH (Chin drives into breast)
        stage.classList.remove('gaping'); // Remove tilt to allow dive
        stage.classList.add('latching'); // Dive in
        
        status.innerHTML = '<span><strong>Step 3:</strong> Asymmetric Latch! Chin first, then lips seal.</span>';
        
        // Show tongue cupping
        setTimeout(() => {
          tongue.style.opacity = '1'; 
          status.className = 'latch-feedback-box success';
          status.innerHTML = '<i class="fas fa-check-circle" style="margin-right:10px;"></i><span><strong>Success!</strong> Deep latch achieved. Milk flowing.</span>';
          spawnMilkParticles();
        }, 600);
        
      }, 1000); // Wait for gape
      
    }, 800); // Wait for align
  }

  function spawnMilkParticles() {
    const stage = document.getElementById('stage3D');
    // Create particles originating from the latch point
    for(let i=0; i<15; i++) {
      let p = document.createElement('div');
      p.className = 'milk-particle';
      // Spray outward from breast
      let x = (Math.random() - 0.5) * 50;
      let y = (Math.random() - 0.5) * 50;
      p.style.transform = `translate(${x}px, ${y}px)`;
      p.style.transition = "all 1s ease-out";
      
      stage.appendChild(p);
      
      setTimeout(() => {
        p.style.opacity = 0;
        p.style.transform = `translate(${x*2}px, ${y*2}px) scale(0)`;
      }, 50);
      
      setTimeout(() => p.remove(), 1000);
    }
  }
</script>

    // Generate theme sections
    function generateThemeSections() {
      const container = document.getElementById('themeSections');
      
      Object.entries(themes).forEach(([themeKey, themeData]) => {
        const section = document.createElement('div');
        section.className = `theme-section ${themeKey === 'physical' ? 'active' : ''}`;
        section.id = `theme-${themeKey}`;
        
        const themeTopics = topics.filter(t => t.theme === themeKey);
        
        section.innerHTML = `
          <div class="theme-header">
            <span class="theme-icon">${themeData.icon}</span>
            <h2 class="theme-title">${themeData.title}</h2>
            <p class="theme-description">${themeData.description}</p>
          </div>
          
          <div class="topic-grid">
            ${themeTopics.map(topic => `
              <div class="topic-card ${themeKey}" data-id="${topic.id}">
                <div class="topic-header">
                  <span class="topic-icon">${topic.icon}</span>
                  <h3 class="topic-title">${topic.title}</h3>
                  <span class="topic-number">Topic ${topic.id}</span>
                </div>
                <div class="topic-body">
                  <p class="topic-preview">${topic.preview}</p>
                  <button class="expand-btn">
                    <i class="fas fa-expand-alt"></i> Explore Details
                  </button>
                </div>
              </div>
            `).join('')}
          </div>
        `;
        
        container.appendChild(section);
      });
      
      // Add click listeners
      document.querySelectorAll('.topic-card').forEach(card => {
        card.addEventListener('click', function() {
          openModal(parseInt(this.dataset.id));
        });
      });
    }

    function switchTheme(themeKey) {
      // Update buttons
      document.querySelectorAll('.theme-btn').forEach(btn => btn.classList.remove('active'));
      document.querySelector(`.theme-btn.${themeKey}`).classList.add('active');
      
      // Update sections
      document.querySelectorAll('.theme-section').forEach(sec => sec.classList.remove('active'));
      document.getElementById(`theme-${themeKey}`).classList.add('active');
    }

    function openModal(id) {
      const topic = topics.find(t => t.id === id);
      const modal = document.getElementById('modal');
      const backdrop = document.getElementById('modalBackdrop');
      const title = document.getElementById('modalTitle');
      const body = document.getElementById('modalBody');
      
      title.innerHTML = `${topic.icon} ${topic.title}`;
      
      const attrNames = {
        definition:"Definition",importance:"Importance",timing:"Timing",
        techniques:"Techniques",signs:"Success Signs",problems:"Common Problems",
        solutions:"Solutions",help:"When to Seek Help",duration:"Duration",
        resources:"Resources",partner:"Partner Role",cultural:"Cultural Notes"
      };
      
      const icons = {
        definition:'info-circle',importance:'star',timing:'clock',techniques:'wrench',
        signs:'check-circle',problems:'exclamation-triangle',solutions:'lightbulb',
        help:'user-md',duration:'hourglass-half',resources:'book',
        partner:'hands-helping',cultural:'globe'
      };
      
      body.innerHTML = Object.entries(topic.attrs).map(([k,v]) => `
        <div class="attr-section">
          <h3><i class="fas fa-${icons[k]}"></i> ${attrNames[k]}</h3>
          <p>${v}</p>
        </div>
      `).join('');
      
      modal.classList.add('active');
      backdrop.classList.add('active');
      document.body.style.overflow = 'hidden';
      
      const card = document.querySelector(`[data-id="${id}"]`);
      if (!card.classList.contains('explored')) {
        card.classList.add('explored');
        updateProgress();
      }
    }

      document.getElementById('modal').classList.remove('active');
      document.getElementById('modalBackdrop').classList.remove('active');
      document.body.style.overflow = '';
    }

    // Body Changes Simulator
    (function() {
      const canvas = document.getElementById('bodyChangesCanvas');
      if (!canvas) return;
      const ctx = canvas.getContext('2d');
      const slider = document.getElementById('weekSlider');
      const weekDisplay = document.getElementById('weekDisplay');
      const changesList = document.getElementById('changesList');
      
      canvas.width = canvas.offsetWidth;
      canvas.height = 400;
      
      function getDataForWeek(week) {
        const keys = Object.keys(bodyChanges).map(Number).sort((a,b) => a-b);
        for (let i = keys.length - 1; i >= 0; i--) {
          if (week >= keys[i]) return bodyChanges[keys[i]];
        }
        return bodyChanges[0];
      }
      
      function draw(week) {
        ctx.clearRect(0, 0, canvas.width, canvas.height);
        
        const cx = canvas.width / 2;
        const cy = canvas.height / 2;
        const baseRadius = 80;
        const growthFactor = 1 + (week / 52) * 0.3;
        const radius = baseRadius * growthFactor;
        
        // Draw breasts
        const spacing = radius * 0.6;
        
        // Left breast
        ctx.beginPath();
        ctx.arc(cx - spacing, cy, radius, 0, Math.PI * 2);
        ctx.fillStyle = 'rgba(236,64,122,0.15)';
        ctx.fill();
        ctx.strokeStyle = 'rgba(236,64,122,0.4)';
        ctx.lineWidth = 3;
        ctx.stroke();
        
        // Right breast
        ctx.beginPath();
        ctx.arc(cx + spacing, cy, radius, 0, Math.PI * 2);
        ctx.fill();
        ctx.stroke();
        
        // Draw milk ducts
        const ductCount = Math.min(8, 4 + Math.floor(week / 8));
        for (let i = 0; i < ductCount; i++) {
          const angle = (Math.PI * 2 / ductCount) * i;
          
          // Left breast ducts
          ctx.beginPath();
          ctx.moveTo(cx - spacing, cy);
          ctx.lineTo(cx - spacing + Math.cos(angle) * radius * 0.7, cy + Math.sin(angle) * radius * 0.7);
          ctx.strokeStyle = `rgba(156,39,176,${0.3 + (week / 52) * 0.3})`;
          ctx.lineWidth = 2;
          ctx.stroke();
          
          // Right breast ducts
          ctx.beginPath();
          ctx.moveTo(cx + spacing, cy);
          ctx.lineTo(cx + spacing + Math.cos(angle) * radius * 0.7, cy + Math.sin(angle) * radius * 0.7);
          ctx.stroke();
        }
        
        // Draw nipples
        ctx.beginPath();
        ctx.arc(cx - spacing, cy, 12, 0, Math.PI * 2);
        ctx.fillStyle = 'rgba(233,30,99,0.5)';
        ctx.fill();
        
        ctx.beginPath();
        ctx.arc(cx + spacing, cy, 12, 0, Math.PI * 2);
        ctx.fill();
        
        // Week indicator
        ctx.fillStyle = '#212121';
        ctx.font = 'bold 16px Inter, sans-serif';
        ctx.textAlign = 'center';
        ctx.fillText(`Week ${week}`, cx, 30);
        
        // Supply indicator
        const supplyLevel = Math.min(100, 20 + (week / 52) * 80);
        ctx.fillStyle = '#555';
        ctx.font = '14px Inter, sans-serif';
        ctx.fillText(`Milk Production: ${supplyLevel.toFixed(0)}%`, cx, canvas.height - 20);
      }
      
      function updateChanges(week) {
        const data = getDataForWeek(week);
        weekDisplay.textContent = data.title;
        
        changesList.innerHTML = data.changes.map(change => `
          <div class="change-item">
            <h4>${change.title}</h4>
            <p>${change.desc}</p>
          </div>
        `).join('');
      }
      
      slider.addEventListener('input', function() {
        const week = parseInt(this.value);
        draw(week);
        updateChanges(week);
      });
      
      draw(0);
      updateChanges(0);
    })();

<script>
  /* =========================================
     BIO-TWIN SIMULATION LOGIC (V3.0)
     ========================================= */
  
  const bioState = {
    running: false,
    timer: null,
    
    // Core Stats (0-100)
    milk: 30,       // Fullness of breasts (visual level)
    capacity: 100,  // Max milk capacity
    hunger: 20,     // Baby hunger
    hydration: 80,  // Mom's water level
    hormones: 50,   // Prolactin level
    energy: 80,     // Mom's fatigue
    stash: 0,       // Stored milk in OZ
    isBusy: false   // Blocking flag
  };

  // --- MAIN LOOP (Runs every 200ms) ---
  function tick() {
    if(!bioState.running) return;

    // 1. Natural Decay
    bioState.hunger = Math.min(100, bioState.hunger + 0.3); 
    bioState.hydration = Math.max(0, bioState.hydration - 0.1); 
    bioState.energy = Math.max(0, bioState.energy - 0.05); 
    bioState.hormones = Math.max(20, bioState.hormones - 0.1); 

    // 2. Milk Production (Rate = Hydration + Hormones)
    // "Refilling" visual
    let rate = 0.2 * (bioState.hydration/100) * (bioState.hormones/50);
    
    if (bioState.milk < 98) {
        bioState.milk = Math.min(bioState.capacity, bioState.milk + rate);
        showBreastStatus("Refilling...", 0.6);
    } else {
        showBreastStatus("Full (FIL Active)", 1); // Feedback Inhibitor of Lactation
    }

    updateUI();
    checkWarnings();
  }

  // --- ACTIONS ---

  function actNurse() {
    if(bioState.isBusy || !bioState.running) return;
    setBusy(true);
    
    // VISUALS
    document.getElementById('babyHead').classList.add('active');
    runBtnProgress('progNurse', 3000);
    showExplain("Nursing", "Direct nipple stimulation releases Oxytocin (Let-down) and Prolactin (Make-more). Most efficient way to feed.");
    updateNarrator("Nursing Started...", "Baby latched. Signaling brain to release milk.", "success");
    showOverlayText("Oxytocin Release!");

    setTimeout(() => {
      // 1.5s in - Letdown happens
      showOverlayText("Let-down Reflex: Milk Flowing");
      
      // Calculate Transfer
      let milkRemoved = Math.min(bioState.milk, 40); // Baby takes chunks
      bioState.milk -= milkRemoved;
      
      // Satisfaction Math (Milk * 3 reduces hunger)
      bioState.hunger = Math.max(0, bioState.hunger - (milkRemoved * 3));
      
      // Boosts
      bioState.hormones = Math.min(100, bioState.hormones + 25);
      bioState.energy = Math.max(0, bioState.energy - 5); // Nursing burns calories

      spawnFloatingText(`-${Math.floor(milkRemoved)}% Vol`, 'milkL', '#ec407a');
    }, 1500);

    setTimeout(() => {
      document.getElementById('babyHead').classList.remove('active');
      
      if(bioState.hunger < 15) {
        updateNarrator("Feed Complete", "Baby is full and sleepy. Hindmilk received.", "normal");
      } else {
        updateNarrator("Still Hungry", "Baby unlatched but needs more. Try switching sides.", "warning");
      }
      hideOverlayText();
      setBusy(false);
    }, 3000);
  }

  function actPump() {
    if(bioState.isBusy || !bioState.running) return;
    setBusy(true);

    // VISUALS
    document.getElementById('pumpKit').classList.add('active');
    document.getElementById('fillL').style.height = '100%';
    document.getElementById('fillR').style.height = '100%';
    runBtnProgress('progPump', 3000);
    showExplain("Pumping", "Uses suction to mimic baby. Emptying breasts removes FIL protein, signaling body to make MORE milk tomorrow.");
    updateNarrator("Pumping...", "Extracting milk for storage.", "normal");
    showOverlayText("Vacuum Suction Active");

    setTimeout(() => {
        showOverlayText("Draining Ducts...");
        // Logic
        let pumpedAmount = bioState.milk; 
        bioState.milk = 0; // Empty completely
        
        // Convert to Stash (approx 10 units = 1 oz for simulation)
        // Ensure minimum pump if breasts had milk
        let ounces = Math.floor(pumpedAmount / 10);
        if (pumpedAmount > 10 && ounces === 0) ounces = 1; 
        
        bioState.stash += ounces;
        bioState.hormones = Math.min(100, bioState.hormones + 15); 

        // Particle Animation flying to stash
        if(ounces > 0) animateMilkToStash();

    }, 1500);

    setTimeout(() => {
      document.getElementById('pumpKit').classList.remove('active');
      document.getElementById('fillL').style.height = '0%';
      document.getElementById('fillR').style.height = '0%';
      hideOverlayText();
      
      updateNarrator("Pumping Done", "Breasts emptied. Feedback loop set to 'Produce'.", "success");
      setBusy(false);
    }, 3000);
  }

  function actHydrate() {
    if(bioState.isBusy || !bioState.running) return;
    
    showExplain("Hydration", "Breastmilk is 87% water. Staying hydrated keeps your blood volume up for efficient milk synthesis.");
    updateNarrator("Gulp Gulp!", "Hydration restored. Production rate increased.", "success");
    
    // Instant Effect
    bioState.hydration = Math.min(100, bioState.hydration + 40);
    spawnFloatingText("+Flow Rate", 'bar-hydro', '#42a5f5');
  }

  function actSleep() {
    if(bioState.isBusy || !bioState.running) return;
    setBusy(true);
    
    document.getElementById('sleepOverlay').classList.add('active');
    runBtnProgress('progSleep', 3000);
    showExplain("Rest", "Cortisol (stress hormone) inhibits milk flow. Sleep lowers cortisol and lets prolactin receptors reset.");
    updateNarrator("Sleeping...", "Time passing... Body recovering.", "normal");

    let ffInterval = setInterval(() => {
        bioState.energy = Math.min(100, bioState.energy + 2);
        bioState.hormones = Math.min(100, bioState.hormones + 0.5);
        // Milk fills up while sleeping
        bioState.milk = Math.min(bioState.capacity, bioState.milk + 1.5);
        // Baby gets hungrier!
        bioState.hunger = Math.min(100, bioState.hunger + 1.5);
    }, 100);

    setTimeout(() => {
      clearInterval(ffInterval);
      document.getElementById('sleepOverlay').classList.remove('active');
      if(bioState.hunger > 80) updateNarrator("Woke Up!", "Rested, but Baby is hungry!", "danger");
      else updateNarrator("Good Morning", "Energy restored.", "normal");
      setBusy(false);
    }, 3000);
  }

    document.getElementById('stashVal').innerText = bioState.stash;

    particle.style.left = (pumpRect.left + 30) + "px";
    particle.style.top = (pumpRect.top + scrollY) + "px";
    
    document.body.appendChild(particle);
    
    // 2. Fly to Stash Box
    setTimeout(() => {
      particle.style.left = (stashRect.left + 50) + "px";
      particle.style.top = (stashRect.top + scrollY + 20) + "px";
      particle.style.transform = "scale(0.5)";
      particle.style.opacity = "0.5";
    }, 50);

    // 3. Land and Pop
    setTimeout(() => {
        document.body.removeChild(particle);
        const stashVal = document.getElementById('stashVal');
        stashVal.classList.remove('stash-pop');
        void stashVal.offsetWidth; // trigger reflow
        stashVal.classList.add('stash-pop');
    }, 1000);
  }

  function spawnFloatingText(text, elementId, color) {
    const el = document.getElementById(elementId);
    if(!el) return;
    const rect = el.getBoundingClientRect();
    const float = document.createElement('div');
    float.className = 'float-text';
    float.innerText = text;
    float.style.color = color;
    // Position relative to parent container ideally, but body works for demo
    float.style.left = (rect.left + window.scrollX + 20) + 'px';
    float.style.top = (rect.top + window.scrollY) + 'px';
    document.body.appendChild(float);
    setTimeout(() => float.remove(), 1500);
  }

  function runBtnProgress(id, duration) {
    const bar = document.getElementById(id);
    bar.style.transition = `width ${duration}ms linear`;
    bar.style.width = '100%';
    setTimeout(() => {
        bar.style.transition = 'none';
        bar.style.width = '0%';
    }, duration);
  }

  function setBar(id, val) { document.getElementById(id).style.width = val + '%'; }
  
  function setBusy(busy) {
    bioState.isBusy = busy;
    document.querySelectorAll('.bio-btn').forEach(b => b.classList.toggle('disabled', busy));
  }

  function updateNarrator(main, sub, type) {
    const bar = document.getElementById('narratorBar');
    document.getElementById('mainStatus').innerHTML = main;
    document.getElementById('subStatus').innerText = sub;
    if(type === 'danger') bar.style.borderLeftColor = '#ef5350';
    else if(type === 'success') bar.style.borderLeftColor = '#66bb6a';
    else if(type === 'warning') bar.style.borderLeftColor = '#ffa726';
    else bar.style.borderLeftColor = '#ec407a';
  }

  function addTag(text, type) {
    const span = document.createElement('span');
    span.className = `status-tag tag-${type}`;
    span.innerText = text;
    document.getElementById('tags').appendChild(span);
  }

  function toggleSim() {
    bioState.running = !bioState.running;
    const icon = document.getElementById('playIcon');
    if(bioState.running) {
      icon.className = "fa-solid fa-pause-circle";
      if(!bioState.timer) bioState.timer = setInterval(tick, 200);
      updateNarrator("Simulation Running", "Monitor stats and take action.", "normal");
    } else {
      icon.className = "fa-solid fa-play-circle";
      updateNarrator("Paused", "Click play to resume.", "normal");
    }
  }
